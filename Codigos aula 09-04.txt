Exercicio 3: 

soma = 0
contador = 0

while True:
    numero = float(input("Digite um numero (ou -1 para sair): "))
    if numero == -1:
        break
    soma += numero
    contador += 1

if contador > 0:
    media = soma / contador
    print(f"Media dos numeros digitados: {media}")
else:
    print("Nenhum numero valido foi digitado")

n = int(input("Digite um numero inteiro N: "))

soma = 0
i = 1
while i <= n:
    soma += i
    i += 1

print(f"O somatorio de 1 ate {n} é {soma}")
    


Exercicio 5: 

pares = 0
impares = 0
contador = 0

while True:
    numero = int(input(f"Digite o {contador + 1}º numero: "))
    
    if numero == (numero // 2) * 2:
        pares += 1
    else:
        impares += 1
    
    contador += 1
    
    if contador == 10:
        break

print(f"\nQuantidade de numeros pares: {pares}")
print(f"Quantidade de numeros impares: {impares}")
